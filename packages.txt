# kind  name  os  spec...
#
#   kind : tap | brew | cask   Homebrew — macOS ONLY, there is no linuxbrew here
#          sys                 distro package (apt / dnf / pacman)
#          gh                  prebuilt release binary — spec is a command
#          sh                  official installer script — spec is a command
#          go                  go install <name>@latest
#          bun                 bun add -g
#   name : what to TEST FOR (gh/sh rows), else the package name. Package and
#          binary names diverge constantly: bottom→btm, git-delta→delta,
#          worktrunk→wt, ripgrep→rg.
#   os   : - | darwin | linux | debian | fedora | arch
#   spec : command to run when `name` is missing. gh rows may call gh_pkg /
#          gh_tar / gh_raw; %v %d %r %m expand to version and deb/rpm/uname arch.
#
# Priority: one universal installer for every OS > distro package > release
# binary. cargo install is deliberately unused — it compiles from source for
# minutes where prebuilt .deb/.rpm/bottles already exist.

# ── universal: identical command on macOS and every Linux ───────────────
sh    bun        -  curl -fsSL https://bun.com/install | bash
sh    claude     -  curl -fsSL https://claude.ai/install.sh | bash
sh    herdr      -  curl -fsSL https://herdr.dev/install.sh | sh
sh    uv         -  curl -LsSf https://astral.sh/uv/install.sh | sh
sh    zoxide     -  curl -sSfL https://raw.githubusercontent.com/ajeetdsouza/zoxide/main/install.sh | sh
sh    starship   -  curl -sS https://starship.rs/install.sh | sh -s -- -y -b "$HOME/.local/bin"
sh    just       -  curl --proto '=https' --tlsv1.2 -sSf https://just.systems/install.sh | bash -s -- --force --to "$HOME/.local/bin"
sh    wt         -  curl --proto '=https' --tlsv1.2 -LsSf https://github.com/max-sixty/worktrunk/releases/latest/download/worktrunk-installer.sh | WORKTRUNK_INSTALL_DIR="$HOME/.local" sh   # it appends /bin

# fd and bat: Debian renames the binaries to fdfind and batcat, which would
# break `alias cat='bat'`. Upstream ships .deb, so take that instead.

# bottom is in no Debian/Fedora repo and its binary is btm. No aarch64 .rpm

gh    btm      fedora  gh_pkg ClementTsang/bottom "bottom-%v-1.%r.rpm" || gh_tar ClementTsang/bottom "bottom_%m-unknown-linux-gnu.tar.gz" btm
