# bottom is NOT packaged by Debian or Fedora, and its binary is `btm`, not
# `bottom`. Upstream ships .deb and .rpm, so install those through the system
# package manager and let the distro own the files. No aarch64 .rpm exists,
# hence the tarball fallback on Fedora.
brew  bottom                          darwin
gh    btm                             debian  gh_pkg ClementTsang/bottom "bottom_%v-1_%d.deb"
gh    btm                             fedora  gh_pkg ClementTsang/bottom "bottom-%v-1.%r.rpm" || gh_tar ClementTsang/bottom "bottom_%m-unknown-linux-gnu.tar.gz" btm
sys   bottom                          arch
